cmake_minimum_required(VERSION 3.15)
project(SEIR_Models LANGUAGES CXX)

set(CMAKE_CXX_STANDARD 11)
set(CMAKE_CXX_STANDARD_REQUIRED True)

if(CMAKE_CXX_COMPILER_ID MATCHES "GNU|Clang")
    add_compile_options(-O3 -Wall)
endif()

find_package(pybind11 REQUIRED)

pybind11_add_module(seir_forward_euler forward_euler/seir_forward_euler.cpp)
set_target_properties(seir_forward_euler PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR})

pybind11_add_module(seir_monte_carlo monte_carlo/seir_monte_carlo.cpp)
set_target_properties(seir_forward_euler PROPERTIES LIBRARY_OUTPUT_DIRECTORY ${CMAKE_SOURCE_DIR})
